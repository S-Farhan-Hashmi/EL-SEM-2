Long readUltrasonic() {

Void checkConfigUpdate() {
If (!Firebase.ready()) return;
If (Firebase.RTDB.getFloat(&fbdo, root_path + “/Config/ThresholdCM”)) {
Float newVal = fbdo.to<float>();
If (newVal > 0 && newVal != threshold_cm) {
Threshold_cm = newVal;
Serial.printf(“Threshold updated: %.2f cm\n”, threshold_cm);
}
}
}

/******** SETUP *********/
Void setup() {
Serial.begin(115200);
pinMode(TRIG_PIN, OUTPUT);
pinMode(ECHO_PIN, INPUT);
pinMode(RESET_PIN, INPUT_PULLUP);

Serial.println(“Hold BOOT to reset…”);
Delay(2000);
If (digitalRead(RESET_PIN) == LOW) {
Prefs.begin(“mod_config”, false);

Prefs.clear();
Prefs.end();
Serial.println(“Config cleared. Restarting…”);
ESP.restart();
}

Prefs.begin(“mod_config”, false);
isConfigured = prefs.getBool(“configured”, false);

if (!isConfigured) {
WiFi.mode(WIFI_STA);
WiFi.begin(SETUP_SSID, SETUP_PASS);
While (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print(“.”); }
Serial.print(“\nOpen this IP to configure: “);
Serial.println(WiFi.localIP());
Server.on(“/”, handleRoot);
Server.on(“/save”, handleSave);
Server.begin();

} else {
Wifi_ssid

= prefs.getString(“ssid”);

Wifi_pass

= prefs.getString(“pass”);

Fb_key

= prefs.getString(“apikey”);

Fb_url

= prefs.getString(“dburl”);

Module_title = prefs.getString(“title”);
Module_lat = prefs.getDouble(“lat”);

Module_lon = prefs.getDouble(“lon”);
Module_uuid = prefs.getString(“uuid”);
Module_id

= prefs.getInt(“id”);

If (prefs.isKey(“thresh”)) threshold_cm = prefs.getFloat(“thresh”);

Root_path = “/stations/” + String(module_id);

WiFi.begin(wifi_ssid.c_str(), wifi_pass.c_str());
While (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print(“.”); }
Serial.println(“\nWiFi connected”);

Config.api_key

= fb_key;

Config.database_url = fb_url;
If (Firebase.signUp(&config, &auth, “”, “”)) {
Serial.println(“Firebase token generated”);
}
Firebase.begin(&config, &auth);
Firebase.reconnectWiFi(true);

uploadStaticInfo();
}
Prefs.end();
}

/******** LOOP *********/
Void loop() {

If (!isConfigured) {
Server.handleClient();
Return;
}

If (millis() – lastConfigCheck > 60000) {
lastConfigCheck = millis();
checkConfigUpdate();
}

If (Firebase.ready() && (millis() – lastSensorRead > 1000)) {
lastSensorRead = millis();

long distance = readUltrasonic();
if (distance == 0) distance = 999;

bool isCarPresent = (distance < threshold_cm);
String newStatus = isCarPresent ? “Occupied” : “Available”;

Serial.printf(“Dist: %ldcm | Status: %s\n”, distance, newStatus.c_str());

String livePath = root_path + “/Live”;
Firebase.RTDB.setFloat(&fbdo, livePath + “/Distance”, (float)distance);

If (newStatus != currentStatus) {
currentStatus = newStatus;

Firebase.RTDB.setString(&fbdo, livePath + “/Status”, newStatus);
}
}
