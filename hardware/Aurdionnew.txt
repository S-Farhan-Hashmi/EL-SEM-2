/******** HELPER FUNCTIONS *********/
String generateUUID() {
  String uuid = "";
  for (int i = 0; i < 32; i++) {
    int r = random(0, 16);
    if (i == 8 || i == 12 || i == 16 || i == 20) uuid += "-";
    uuid += String(r, HEX);
  }
  return uuid;
}

long readUltrasonic() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH);
  return (duration * 0.034 / 2.0);
}
